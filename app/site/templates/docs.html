from aice import Client          # pip install ./sdk/python
c = Client("{{ p.base }}", api_key="sk_…")
job = c.jobs.create({
    "in":  {"operation": "import/url", "url": "https://example.com/invoice.docx"},
    "pdf": {"operation": "convert", "input": "in", "output_format": "pdf"},
    "out": {"operation": "export/url", "input": "pdf"},
})
job = c.jobs.wait(job["id"])
for f in c.jobs.files(job):
    c.download(f["url"], f["filename"])
